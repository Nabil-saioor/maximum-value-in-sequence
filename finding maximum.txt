My program in Python for: finding the maximum value in a sequence of integers.

n = input("Enter the number of values: ")
n = int(n)
values = []
for i in range(1,n+1):
    value = input(" Enter the value number "+str(i)+":  ")
    value = int(value)
    values.append(value)

print(" Your sequense is: "+str(values)+".")
maximum = values[0]
for i in range(1,n):
	if values[i] > maximum:
		maximum = values[i]

print(" \nThe maximum value is: "+str(maximum)+".")
